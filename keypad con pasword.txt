#include <Key.h>
#include <Keypad.h>

const byte FILAS =4;
const byte COLUMNAS =4;
char keys[FILAS][COLUMNAS]={
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};
byte pinesFilas[FILAS]= {9,8,7,6};
byte pinesColumnas[COLUMNAS]={5,4,3,2};

Keypad teclado= Keypad(makeKeymap(keys),pinesFilas, pinesColumnas, FILAS ,COLUMNAS);
char TECLA;
char clave [5];
char clave_original[5]="1234";
byte indice = 0;
byte intentos =0;

void setup (){
   Serial.begin(9600);
   pinMode(13,OUTPUT);
   pinMode(12,OUTPUT);
}
void loop(){

   TECLA= teclado.getKey();
  if(TECLA){
    clave[indice]=TECLA;
    indice ++;
      Serial. print (TECLA);
   }
   if(indice==4){
  if(!strcmp(clave, clave_original)){
  Serial.println ("Contraseña Coreccta"); 
  intentos =0;
  digitalWrite(13,0);   
} else{
  Serial.println ("Contraseña Incoreccta");
  digitalWrite (13,1);
  intentos++;
}
indice=0;
   }
   if(intentos==3){
    digitalWrite(12,1);
    delay(10000);
    digitalWrite(12,0);
    intentos=0;
   }
}
