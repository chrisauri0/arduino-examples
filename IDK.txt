int IN1=2;
int IN2=3;
int ENA=5;

 void setup(){
  pinMode (IN1, OUTPUT);
  pinMode (IN2, OUTPUT);
  pinMode (ENA, OUTPUT);
 }

void loop(){
  digitalWrite (ENA,1);
  digitalWrite (IN1, 0);
  digitalWrite (IN2,1);
  delay (5000);
  digitalWrite (ENA,0);
  delay (2000);
  digitalWrite (ENA, 1);
  digitalWrite (IN1,1);
  digitalWrite(IN2,0);
  delay (5000);
  digitalWrite (ENA,0);
  delay (2000);
}
