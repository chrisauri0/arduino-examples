int IN1 = 2;
int IN2 = 3;
int ENA = 5;
int VELOCIDAD;

void setup (){
  analogWrite (IN1, VELOCIDAD);
  pinMode(IN2, OUTPUT);
  pinMode(ENA, OUTPUT);
}

void loop (){
  for(VELOCIDAD=0;VELOCIDAD<100;VELOCIDAD++){
    digitalWrite(ENA, 1);
    digitalWrite(IN1, 0);
    digitalWrite(IN2, 1);
    delay(50);
    
  }
  digitalWrite(ENA, 0);
  delay(2000);

  for(VELOCIDAD=0;VELOCIDAD<100;VELOCIDAD++){
    digitalWrite(ENA, 1);
    digitalWrite(IN1, 1);
    digitalWrite(IN2, 0);
    delay(50);
    
  }
  digitalWrite(ENA, 0);
  delay(2000);
