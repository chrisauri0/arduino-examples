#include <SoftwareSerial.h>

SoftwareSerial bt (10,11);
int sensor=A0;
int temp;
float temperatura;

void setup(){
  bt.begin(38400);
  Serial.begin(9600);
}
void loop(){
  temp=analogRead(sensor);
  temperatura=(((temp*5000.0)/1023)/10);
  bt.println(temperatura);
  bt.println("C");
  Serial.println("temperatura");
  delay(1000);
}	|
